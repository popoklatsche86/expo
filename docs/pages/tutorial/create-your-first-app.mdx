import 'package:flutter/material.dart';
import 'package:provider/provider.dart';

void main() {
  runApp(MiniRoyaleApp());
}

class MiniRoyaleApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return ChangeNotifierProvider(
      create: (_) => GameState(),
      child: MaterialApp(
        debugShowCheckedModeBanner: false,
        home: GameScreen(),
      ),
    );
  }
}

class GameState extends ChangeNotifier {
  int playerHealth = 100;
  int enemyHealth = 100;

  void attackEnemy() {
    enemyHealth -= 10;
    if (enemyHealth < 0) enemyHealth = 0;
    notifyListeners();
  }

  void enemyAttacks() {
    playerHealth -= 10;
    if (playerHealth < 0) playerHealth = 0;
    notifyListeners();
  }

  void resetGame() {
    playerHealth = 100;
    enemyHealth = 100;
    notifyListeners();
  }
}

class GameScreen extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    final game = Provider.of<GameState>(context);

    return Scaffold(
      appBar: AppBar(title: Text('Mini Royale')),
      body: Column(
        mainAxisAlignment: MainAxisAlignment.center,
        children: [
          Text("🧙 Gegner: ${game.enemyHealth} HP", style: TextStyle(fontSize: 22)),
          SizedBox(height: 20),
          ElevatedButton(
            onPressed: () {
              game.attackEnemy();
              game.enemyAttacks();
            },
            child: Text("🃏 Karte ausspielen"),
          ),
          SizedBox(height: 20),
          Text("🛡️ Du: ${game.playerHealth} HP", style: TextStyle(fontSize: 22)),
          SizedBox(height: 40),
          ElevatedButton(
            onPressed: game.resetGame,
            child: Text("🔁 Neustarten"),
          ),
        ],
      ),
    );
  }
}
